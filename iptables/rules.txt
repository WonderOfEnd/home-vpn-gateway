# Актуальный набор правил (получен через `iptables -t <table> -S`),
# совместим с iptables-restore.
# Перед применением заменить <VPS_IP>. Автоматически созданные цепочки/правила
# Docker (DOCKER, DOCKER-USER, DOCKER-ISOLATION-*, MASQUERADE для 172.17.0.0/16)
# оставлены как есть, так как сосуществуют на этой машине.
#
# Применение, например:
#   sudo iptables-restore --noflush < rules.txt
#
# QUIC (UDP 443) с LAN наружу режется, чтобы браузеры мгновенно откатывались на TCP
# (который идёт через туннель). Иначе QUIC уходит напрямую с реальным IP, и сайты
# с гео-проверкой (например, Claude) блокируют.

# --- filter ---
-P INPUT ACCEPT
-P FORWARD ACCEPT
-P OUTPUT ACCEPT
-N DOCKER
-N DOCKER-ISOLATION-STAGE-1
-N DOCKER-ISOLATION-STAGE-2
-N DOCKER-USER
-A FORWARD -i enp2s0 -o enp2s0 -p udp -m udp --dport 443 -j REJECT --reject-with icmp-port-unreachable
-A FORWARD -j DOCKER-USER
-A FORWARD -j DOCKER-ISOLATION-STAGE-1
-A FORWARD -o docker0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT
-A FORWARD -o docker0 -j DOCKER
-A FORWARD -i docker0 ! -o docker0 -j ACCEPT
-A FORWARD -i docker0 -o docker0 -j ACCEPT
-A FORWARD -i enp2s0 -j ACCEPT
-A FORWARD -o enp2s0 -j ACCEPT
-A FORWARD -i enp2s0 -j ACCEPT
-A FORWARD -o enp2s0 -m state --state RELATED,ESTABLISHED -j ACCEPT
-A FORWARD -i enp2s0 -j ACCEPT
-A FORWARD -o enp2s0 -j ACCEPT
-A DOCKER-ISOLATION-STAGE-1 -i docker0 ! -o docker0 -j DOCKER-ISOLATION-STAGE-2
-A DOCKER-ISOLATION-STAGE-1 -j RETURN
-A DOCKER-ISOLATION-STAGE-2 -o docker0 -j DROP
-A DOCKER-ISOLATION-STAGE-2 -j RETURN
-A DOCKER-USER -j RETURN

# --- nat ---
-P PREROUTING ACCEPT
-P INPUT ACCEPT
-P OUTPUT ACCEPT
-P POSTROUTING ACCEPT
-N DOCKER
-N XRAY
-A PREROUTING -i enp2s0 -p udp -m udp --dport 53 -j REDIRECT --to-ports 53
-A PREROUTING -i enp2s0 -p tcp -m tcp --dport 53 -j REDIRECT --to-ports 53
-A PREROUTING -i enp2s0 -p tcp -j XRAY
-A PREROUTING -m addrtype --dst-type LOCAL -j DOCKER
-A OUTPUT ! -d 127.0.0.0/8 -m addrtype --dst-type LOCAL -j DOCKER
-A POSTROUTING -s 172.17.0.0/16 ! -o docker0 -j MASQUERADE
-A POSTROUTING -o enp2s0 -j MASQUERADE
-A POSTROUTING -o enp2s0 -j MASQUERADE
-A POSTROUTING -o enp2s0 -j MASQUERADE
-A DOCKER -i docker0 -j RETURN
-A XRAY -d 0.0.0.0/8 -j RETURN
-A XRAY -d 10.0.0.0/8 -j RETURN
-A XRAY -d 127.0.0.0/8 -j RETURN
-A XRAY -d 169.254.0.0/16 -j RETURN
-A XRAY -d 172.16.0.0/12 -j RETURN
-A XRAY -d 192.168.0.0/16 -j RETURN
-A XRAY -d 224.0.0.0/4 -j RETURN
-A XRAY -d 240.0.0.0/4 -j RETURN
-A XRAY -d <VPS_IP>/32 -j RETURN
-A XRAY -p tcp -j REDIRECT --to-ports 12345
