# Актуальный набор правил (получен через `iptables -t <table> -S`),
# совместим с iptables-restore.
# Перед применением заменить <VPS_IP>. Автоматически созданные цепочки/правила
# Docker (DOCKER, DOCKER-USER, DOCKER-ISOLATION-*, MASQUERADE для 172.17.0.0/16)
# оставлены как есть, так как сосуществуют на этой машине — можно игнорировать
# или убрать, если Docker/Pi-hole у вас не используется.
#
# Применение, например:
#   sudo iptables-restore --noflush < rules.txt
# (или по таблицам отдельно, `iptables -t <table> -S` куда нужно)

# --- filter ---
-P INPUT ACCEPT
-P FORWARD ACCEPT
-P OUTPUT ACCEPT
-N DOCKER
-N DOCKER-ISOLATION-STAGE-1
-N DOCKER-ISOLATION-STAGE-2
-N DOCKER-USER
-A FORWARD -j DOCKER-USER
-A FORWARD -j DOCKER-ISOLATION-STAGE-1
-A FORWARD -o docker0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT
-A FORWARD -o docker0 -j DOCKER
-A FORWARD -i docker0 ! -o docker0 -j ACCEPT
-A FORWARD -i docker0 -o docker0 -j ACCEPT
-A FORWARD -i enp2s0 -j ACCEPT
-A FORWARD -o enp2s0 -j ACCEPT
-A FORWARD -o enp2s0 -m state --state RELATED,ESTABLISHED -j ACCEPT
-A DOCKER-ISOLATION-STAGE-1 -i docker0 ! -o docker0 -j DOCKER-ISOLATION-STAGE-2
-A DOCKER-ISOLATION-STAGE-1 -j RETURN
-A DOCKER-ISOLATION-STAGE-2 -o docker0 -j DROP
-A DOCKER-ISOLATION-STAGE-2 -j RETURN
-A DOCKER-USER -j RETURN

# --- nat ---
# Редиректит исходящий из локальной сети DNS (на любой адрес назначения) на
# локальный резолвер (Pi-hole), редиректит исходящий TCP на прозрачный
# инбаунд xray (кроме диапазонов-исключений ниже), и NAT'ит всё остальное
# наружу через enp2s0.
-P PREROUTING ACCEPT
-P INPUT ACCEPT
-P OUTPUT ACCEPT
-P POSTROUTING ACCEPT
-N DOCKER
-N XRAY
-A PREROUTING -i enp2s0 -p udp -m udp --dport 53 -j REDIRECT --to-ports 53
-A PREROUTING -i enp2s0 -p tcp -m tcp --dport 53 -j REDIRECT --to-ports 53
-A PREROUTING -i enp2s0 -p tcp -j XRAY
-A PREROUTING -m addrtype --dst-type LOCAL -j DOCKER
-A OUTPUT ! -d 127.0.0.0/8 -m addrtype --dst-type LOCAL -j DOCKER
-A POSTROUTING -s 172.17.0.0/16 ! -o docker0 -j MASQUERADE
-A POSTROUTING -o enp2s0 -j MASQUERADE
-A DOCKER -i docker0 -j RETURN
-A XRAY -d 0.0.0.0/8 -j RETURN
-A XRAY -d 10.0.0.0/8 -j RETURN
-A XRAY -d 127.0.0.0/8 -j RETURN
-A XRAY -d 169.254.0.0/16 -j RETURN
-A XRAY -d 172.16.0.0/12 -j RETURN
-A XRAY -d 192.168.0.0/16 -j RETURN
-A XRAY -d 224.0.0.0/4 -j RETURN
-A XRAY -d 240.0.0.0/4 -j RETURN
-A XRAY -d <VPS_IP>/32 -j RETURN
-A XRAY -p tcp -j REDIRECT --to-ports 12345

# --- mangle ---
# Цепочка XRAY_UDP определена, но сейчас НЕ подключена к PREROUTING — см.
# раздел "Почему UDP не туннелируется" в README.md. Оставлена как есть, на
# случай если полный UDP TPROXY когда-нибудь снова понадобится включить
# (добавить хук в PREROUTING обратно, см. ниже).
-P PREROUTING ACCEPT
-P INPUT ACCEPT
-P FORWARD ACCEPT
-P OUTPUT ACCEPT
-P POSTROUTING ACCEPT
-N XRAY_UDP
-A XRAY_UDP -d 0.0.0.0/8 -j RETURN
-A XRAY_UDP -d 10.0.0.0/8 -j RETURN
-A XRAY_UDP -d 127.0.0.0/8 -j RETURN
-A XRAY_UDP -d 169.254.0.0/16 -j RETURN
-A XRAY_UDP -d 172.16.0.0/12 -j RETURN
-A XRAY_UDP -d 192.168.0.0/16 -j RETURN
-A XRAY_UDP -d 224.0.0.0/4 -j RETURN
-A XRAY_UDP -d 240.0.0.0/4 -j RETURN
-A XRAY_UDP -d <VPS_IP>/32 -j RETURN
-A XRAY_UDP -p udp -m udp --dport 53 -j RETURN
-A XRAY_UDP -p udp -m udp --dport 19300:19309 -j RETURN
-A XRAY_UDP -p udp -j TPROXY --on-port 12346 --on-ip 127.0.0.1 --tproxy-mark 0x1/0x1
# Чтобы снова включить полный UDP TPROXY (сначала прочитать оговорки в README):
#   iptables -t mangle -A PREROUTING -i enp2s0 -p udp -j XRAY_UDP
#   ip rule add fwmark 0x1 lookup 100
#   ip route add local 0.0.0.0/0 dev lo table 100
